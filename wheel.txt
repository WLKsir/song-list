<script>
const wheelDom = document.getElementById("wheel");
const startBtn = document.getElementById("start-btn");
const resultDom = document.getElementById("result");
const tipDom = document.getElementById("list-tip");

let songList = [];
let rotating = false;
let lastDeg = 0;
const MAX_CAPACITY = 1000;
const SCALE_THRESHOLD = 20;
const HIDE_TEXT_THRESHOLD = 40;
const CX = 210;
const CY = 210;
const OUTER_RADIUS = 204;
const SECTOR_RADIUS = 192;

function init(){
    const str = localStorage.getItem("wheelSongList");
    if(!str){
        tipDom.innerText = "暂无歌曲，请先前往歌单页面筛选！";
        startBtn.disabled = true;
        return;
    }
    songList = JSON.parse(str);
    if(songList.length ===0){
        tipDom.innerText = "筛选结果为空！";
        startBtn.disabled = true;
        return;
    }
    if(songList.length > MAX_CAPACITY){
        alert(`歌曲数量超过上限${MAX_CAPACITY}首，仅加载前${MAX_CAPACITY}首`);
        songList = songList.slice(0, MAX_CAPACITY);
    }
    tipDom.innerText = `转盘载入歌曲：${songList.length}首`;
    drawWheel();
}

function drawWheel(){
    const total = songList.length;
    const sliceDeg = 360 / total;
    const colors = ["#ff6b6b","#4ecdc4","#45b7d1","#ffe066","#a8e6cf","#ffd3a5","#c9b1ff","#ffacc5"];
    let html = `<svg width="420" height="420" viewBox="0 0 420 420">`;
    let fontSize = 12;
    if(total > SCALE_THRESHOLD) fontSize = Math.max(5, 12 - (total - SCALE_THRESHOLD)*0.3);
    const showText = total <= HIDE_TEXT_THRESHOLD;
    for(let i=0;i<total;i++){
        const start = sliceDeg*i;
        const end = sliceDeg*(i+1);
        const color = colors[i % colors.length];
        html += createSlice(start, end, color, songList[i]["歌曲名称"], fontSize, showText);
    }
    html += `<circle cx="${CX}" cy="${CY}" r="${OUTER_RADIUS}" fill="none" stroke="#ffffff" stroke-width="12"/>`;
    html += `</svg>`;
    wheelDom.innerHTML = html;
}

function createSlice(startDeg, endDeg, fill, text, fontSize, showText){
    const deg2rad = d => d * Math.PI / 180;
    const offset = -90;
    const startRad = deg2rad(startDeg + offset);
    const endRad = deg2rad(endDeg + offset);
    const x1 = CX + SECTOR_RADIUS * Math.cos(startRad);
    const y1 = CY + SECTOR_RADIUS * Math.sin(startRad);
    const x2 = CX + SECTOR_RADIUS * Math.cos(endRad);
    const y2 = CY + SECTOR_RADIUS * Math.sin(endRad);
    const largeArc = (endDeg - startDeg) > 180 ? 1 : 0;
    const path = `<path d="M${CX},${CY} L${x1},${y1} A${SECTOR_RADIUS},${SECTOR_RADIUS} 0 ${largeArc},1 ${x2},${y2} Z" fill="${fill}" stroke="#fff" stroke-width="0.8"/>`;
    if(!showText) return path;
    const midDeg = startDeg + (endDeg - startDeg)/2;
    const midRad = deg2rad(midDeg + offset);
    const tx = CX + SECTOR_RADIUS * 0.62 * Math.cos(midRad);
    const ty = CY + SECTOR_RADIUS * 0.62 * Math.sin(midRad);
    return path + `<text x="${tx}" y="${ty}" fill="#000" font-size="${fontSize}" text-anchor="middle" transform="rotate(${midDeg},${tx},${ty})">${text}</text>`;
}

startBtn.onclick = function(){
    if(rotating) return;
    rotating = true;
    startBtn.disabled = true;
    resultDom.style.opacity = "0";
    resultDom.innerText = "";
    const total = songList.length;
    const sliceDeg = 360 / total;
    //随机旋转总圈数
    const randomRotate = 360 * (6 + Math.random() * 5);
    const finalRotate = lastDeg + randomRotate;
    wheelDom.style.transform = `rotate(${finalRotate}deg)`;
    lastDeg = finalRotate;

    setTimeout(()=>{
        rotating = false;
        startBtn.disabled = false;
        // 核心：根据当前旋转角度，实时计算指针指向哪一块扇形
        let currentAngle = lastDeg % 360;
        // 指针固定在顶部0度位置，转盘顺时针旋转，角度反向换算
        let pointerAngle = (360 - currentAngle + 360) % 360;
        let targetIndex = Math.floor(pointerAngle / sliceDeg);
        //边界修正
        if(targetIndex >= total) targetIndex = total -1;
        const pick = songList[targetIndex];
        resultDom.innerText = `选中歌曲：${pick["歌曲名称"]} | ${pick["歌曲版本"]}`;
        resultDom.style.opacity = "1";
    }, 4000)
}
init();
</script>
